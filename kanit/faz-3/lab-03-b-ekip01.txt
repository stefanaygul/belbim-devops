### Adim 4: compose.yaml ve nginx.conf
rol: ekip01-dev
$ cat deploy/compose.yaml
# deploy/compose.yaml - Saha Ekipman Talep Portalı
#
# IMG, TAG ve WEB_PORT ortamdan ya da bu dosyanın yanındaki .env dosyasından gelir:
#   IMG=<registry>/<grup>/<ekip>/saha-ekipman-portal   TAG=1.0.1   WEB_PORT=8081
# Yalnızca web-ui port yayınlar; API'lere Nginx üzerinden /api ile ulaşılır.
services:
  product-service:
    image: ${IMG}/product-service:${TAG}

  order-service:
    image: ${IMG}/order-service:${TAG}
    environment:
      ConnectionStrings__Orders: Data Source=/data/orders.db
    volumes:
      - orders-data:/data

  web-ui:
    image: ${IMG}/web-ui:${TAG}
    ports: ["${WEB_PORT:-8080}:8080"]
    depends_on: [product-service, order-service]

volumes:
  orders-data:
$ cat deploy/nginx.conf
# deploy/nginx.conf - web-ui imajındaki Nginx ayarı
#
# Tarayıcı yalnızca web-ui ile konuşur. /api istekleri Compose servis adlarıyla
# ilgili API'ye yönlendirilir; istek yolu olduğu gibi iletilir.
server {
  listen 8080;
  root /usr/share/nginx/html;

  location / { try_files $uri /index.html; }

  location /api/products {
    proxy_pass http://product-service:8080;
  }

  location /api/orders {
    proxy_pass http://order-service:8080;
  }
}
